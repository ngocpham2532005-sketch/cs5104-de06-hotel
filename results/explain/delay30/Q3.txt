FROM (
  SELECT booking_id, region, hotel, room_id, room_type_code, check_in, nights, guest_name FROM s_dalat.v_tour_room WHERE tour_group_id = 410
  UNION ALL SELECT booking_id, region, hotel, room_id, room_type_code, check_in, nights, guest_name FROM s_nhatrang.v_tour_room WHERE tour_group_id = 410
  UNION ALL SELECT booking_id, region, hotel, room_id, room_type_code, check_in, nights, guest_name FROM s_phuquoc.v_tour_room WHERE tour_group_id = 410
  UNION ALL SELECT booking_id, region, hotel, room_id, room_type_code, check_in, nights, guest_name FROM s_danang.v_tour_room WHERE tour_group_id = 410
) x
ORDER BY check_in, booking_id;
